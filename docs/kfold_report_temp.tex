\subsection{1. Tổng quan các mô
hình}\label{tux1ed5ng-quan-cuxe1c-muxf4-huxecnh}

\subsubsection{1.1. Mô hình v25
(ResNet1D)}\label{muxf4-huxecnh-v25-resnet1d}

\begin{itemize}
\tightlist
\item
  \textbf{Kiến trúc chính:} Là mạng Residual Network 1D. Cấu trúc cốt
  lõi sử dụng các block \texttt{SeparableConv1D} kết hợp kết nối dư
  (skip-connections) và có tích hợp khối SE (Squeeze-and-Excitation).
\item
  \textbf{Kích thước mô hình (Keras):} \textasciitilde502 KB.
\item
  \textbf{Đặc điểm \& Ứng dụng:} Là sự cân bằng giữa khả năng trích xuất
  đặc trưng sâu (ResNet) và việc giảm thiểu tham số (SeparableConv).
  Khối SE có hàm \texttt{sigmoid} có thể làm tăng nhẹ thời gian suy luận
  so với v30, nhưng cải thiện đáng kể khả năng nhận diện các nhãn khó
  (như Trans).
\end{itemize}

\subsubsection{1.2. Mô hình v30 (CNN thuần tối ưu
ESP-NN)}\label{muxf4-huxecnh-v30-cnn-thuux1ea7n-tux1ed1i-ux1b0u-esp-nn}

\begin{itemize}
\tightlist
\item
  \textbf{Kiến trúc chính:} Sử dụng cấu trúc mạng Convolutional Neural
  Network 1D (CNN thuần) với các lớp \texttt{Conv1D} và
  \texttt{SeparableConv1D}. Các lớp kích hoạt là \texttt{relu6} kết hợp
  với \texttt{BatchNormalization}. Đặc biệt sử dụng cả
  \texttt{GlobalAveragePooling1D} và \texttt{MaxPooling1D} ở đoạn cuối
  để không làm loãng đỉnh gia tốc của các cử động đột ngột (như té ngã).
\item
  \textbf{Kích thước mô hình (Keras):} \textasciitilde174 KB.
\item
  \textbf{Đặc điểm \& Ứng dụng:} Được thiết kế hoàn toàn dựa trên các
  toán tử tối ưu hóa cứng bởi vi xử lý \textbf{ESP-NN} trên chip
  ESP32-S3. Mô hình có khả năng chạy suy luận nhanh (latency cực thấp)
  và tiêu tốn ít bộ nhớ flash.
\end{itemize}

\subsubsection{1.3. Mô hình v30\_tcn (TCN - Temporal Convolutional
Network)}\label{muxf4-huxecnh-v30_tcn-tcn---temporal-convolutional-network}

\begin{itemize}
\tightlist
\item
  \textbf{Kiến trúc chính:} Dựa trên các lớp \texttt{Conv1D} với hệ số
  giãn nở (Dilation Rate: 1, 2, 4, 8) nhằm gia tăng kích thước cửa sổ
  nhìn (Receptive Field) để bắt được ngữ cảnh chuỗi thời gian dài. Ngoài
  ra, mô hình còn sử dụng cấu trúc Residual (Skip-connections) kết hợp
  khối SE (Squeeze-and-Excitation) dùng hàm \texttt{sigmoid}.
\item
  \textbf{Kích thước mô hình (Keras):} \textasciitilde975 KB.
\item
  \textbf{Đặc điểm \& Ứng dụng:} TCN ghi nhớ ngữ cảnh chuỗi thời gian
  vượt trội, đem lại hiệu suất phân loại xuất sắc. Tuy nhiên,
  \textbf{dilated convolution và sigmoid không được ESP-NN hỗ trợ tăng
  tốc}. Mạch ESP32-S3 sẽ phải dùng reference kernel để chạy, dẫn đến độ
  trễ suy luận rất cao (\textasciitilde100ms hoặc hơn).
\end{itemize}

\subsection{3. Đánh giá chuyên sâu và Nhận
xét}\label{ux111uxe1nh-giuxe1-chuyuxean-suxe2u-vuxe0-nhux1eadn-xuxe9t}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  \textbf{Hiệu suất tổng thể (Accuracy \& Macro F1):} Mô hình
  \textbf{v30\_tcn (TCN)} đạt độ chính xác và Macro F1 tốt nhất trên hầu
  hết các kịch bản đánh giá. Nó đặc biệt vượt trội ở các kịch bản dự
  đoán chéo khó nhằn như S3 (Train trên người già, Test trên người trẻ),
  cho thấy khả năng tổng quát hóa xuất sắc.
\item
  \textbf{Fall Recall (Chỉ số ưu tiên số 1):} Trên tập S5 (trộn chung dữ
  liệu), cả 3 mô hình đều đạt Fall Recall rất cao (\textgreater99.2\%).
  Điểm tạo nên sự khác biệt lớn nhất là ở kịch bản S3. \textbf{v30\_tcn}
  giữ được mức Recall ấn tượng là \textbf{95.86\%}, trong khi v25
  (ResNet) đạt 88.11\% và v30 (CNN) sụt giảm mạnh xuống 85.86\%. TCN thể
  hiện sức chống chịu (robustness) rất tốt trước sự thay đổi miền dữ
  liệu.
\item
  \textbf{Phân loại nhãn dễ nhầm lẫn - Trans (Chuyển tiếp):}
  \textbf{v30\_tcn} xử lý cực kỳ xuất sắc việc phân loại nhãn Trans. Cụ
  thể trên tập S5, Trans F1 của TCN đạt \textbf{93.71\%}, đánh bại hoàn
  toàn v25 (89.00\%) và v30 (81.85\%). Khả năng nắm bắt chuỗi thời gian
  qua Dilated Convolutions giúp TCN giảm thiểu nhầm lẫn giữa các hoạt
  động phức tạp.
\item
  \textbf{Lưu ý triển khai thực tế (Hardware / ESP-NN Constraints):} Mặc
  dù TCN có độ chính xác tuyệt vời, cấu trúc của nó (Dilated Conv,
  Sigmoid) \textbf{không được bộ vi xử lý ESP-NN hỗ trợ tăng tốc}. Do
  đó, khi nén INT8 và chạy trên chip ESP32-S3, thời gian suy luận
  (inference latency) sẽ rất chậm (\textasciitilde100ms hoặc hơn). Ngược
  lại, \textbf{v25 (ResNet1D)} hoặc \textbf{v30 (CNN thuần)} sử dụng cấu
  trúc cực kỳ thân thiện với vi mạch (pointwise, depthwise conv, relu6),
  mang lại độ trễ siêu thấp (\textasciitilde10ms). Việc chọn mô hình
  cuối cùng sẽ là bài toán đánh đổi (trade-off) giữa độ chính xác và tốc
  độ.
\end{enumerate}
